{{- /* figure shortcode, same parameters as Hugo's built-in one (src, link, target, rel, title, caption, class, alt, attr, attrlink, loading) but the image goes through partials/responsive-image.html when it is a page-bundle or assets/ resource. {{< figure src="bg.jpg" alt="…" caption="…" attr="Source: X" attrlink="https://…" >}} */ -}} {{- $src := .Get "src" -}} {{- $img := "" -}} {{- with .Page.Resources.Get $src -}} {{- $img = . -}} {{- else -}} {{- with resources.Get (strings.TrimPrefix "/" $src) }}{{ $img = . }}{{ end -}} {{- end -}} {{- $alt := .Get "alt" | default (.Get "caption" | markdownify | plainify) -}} {{- with .Get "link" }}{{ end -}} {{ partial "responsive-image.html" (dict "img" $img "src" $src "alt" $alt "title" (.Get "title") "loading" (.Get "loading" | default "lazy")) }} {{- if .Get "link" }}{{ end -}} {{- if or (.Get "title") (.Get "caption") (.Get "attr") }} {{- end }}
