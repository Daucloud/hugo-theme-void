{{- /* Per-page UI string lookup. The site is single-language, but individual posts may declare `language = "en"` in front matter. This partial returns the UI string for the *page's* language: 1. data/ui.toml -> [] 2. fallback to the site-language i18n string @example: {{ partial "ui.html" (dict "page" . "key" "reading_time") }} */ -}} {{- $page := .page -}} {{- $key := .key -}} {{- $lang := $page.Params.language | default site.Language.Lang -}} {{- $lang = index (split (lower (printf "%s" $lang)) "-") 0 -}} {{- $s := "" -}} {{- with index site.Data.ui $lang -}} {{- with index . $key -}}{{- $s = . -}}{{- end -}} {{- end -}} {{- if not $s -}}{{- $s = i18n $key -}}{{- end -}} {{- $s -}}
